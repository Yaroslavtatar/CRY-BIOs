# Coolify → CRY BIOS → Advanced → Labels (по одной строке)
# Домены в Coolify: ТОЛЬКО https://cbios.ru  (НЕ добавляй https://*.cbios.ru — ломает Traefik 3)
# DNS Cloudflare: A cbios.ru + A *.cbios.ru → IP VPS
# Wildcard SSL: deploy/coolify/SSL_SETUP.md (DNS-challenge обязателен)

traefik.enable=true
traefik.docker.network=coolify
traefik.http.services.crybios.loadbalancer.server.port=3000
traefik.http.routers.crybios-http.entrypoints=http
traefik.http.routers.crybios-http.middlewares=redirect-to-https
traefik.http.routers.crybios-http.rule=Host(`cbios.ru`) || HostRegexp(`^[a-z0-9-]+\.cbios\.ru$`)
traefik.http.routers.crybios-http.service=crybios
traefik.http.routers.crybios-apex.entrypoints=https
traefik.http.middlewares.crybios-buffering.buffering.maxRequestBodyBytes=524288000
traefik.http.middlewares.crybios-buffering.buffering.memRequestBodyBytes=20971520
traefik.http.routers.crybios-apex.middlewares=gzip,crybios-buffering
traefik.http.routers.crybios-wildcard.middlewares=gzip,crybios-buffering
traefik.http.routers.crybios-apex.rule=Host(`cbios.ru`) || Host(`www.cbios.ru`)
traefik.http.routers.crybios-apex.service=crybios
traefik.http.routers.crybios-apex.tls=true
traefik.http.routers.crybios-apex.tls.certresolver=letsencrypt
traefik.http.routers.crybios-wildcard.entrypoints=https
traefik.http.routers.crybios-wildcard.rule=HostRegexp(`^[a-z0-9-]+\.cbios\.ru$`)
traefik.http.routers.crybios-wildcard.service=crybios
traefik.http.routers.crybios-wildcard.tls=true
traefik.http.routers.crybios-wildcard.tls.certresolver=letsencrypt
traefik.http.routers.crybios-wildcard.tls.domains[0].main=cbios.ru
traefik.http.routers.crybios-wildcard.tls.domains[0].sans=*.cbios.ru
traefik.http.routers.crybios-wildcard.priority=10
